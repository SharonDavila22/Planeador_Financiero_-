-- ============================
--   TABLAS DE CATALOGO
-- ============================

CREATE TABLE categoria_egreso (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT
);

CREATE TABLE categoria_ingreso (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT
);

CREATE TABLE fondo_inversion (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    tipo VARCHAR(50),
    descripcion TEXT
);

-- ============================
--   PRESUPUESTO MENSUAL
-- ============================

CREATE TABLE presupuesto_mensual (
    id SERIAL PRIMARY KEY,
    anio INT NOT NULL,
    mes INT NOT NULL CHECK (mes BETWEEN 1 AND 12),
    monto NUMERIC(12,2) NOT NULL,
    UNIQUE(anio, mes)
);

-- ============================
--   EGRESOS
-- ============================

CREATE TABLE egreso (
    id SERIAL PRIMARY KEY,
    fecha DATE NOT NULL,
    monto NUMERIC(12,2) NOT NULL,
    categoria_id INT NOT NULL REFERENCES categoria_egreso(id),
    presupuesto_id INT NOT NULL REFERENCES presupuesto_mensual(id),
    descripcion TEXT
);

-- ============================
--   FACTURAS DE EGRESO
-- ============================

CREATE TABLE factura (
    id SERIAL PRIMARY KEY,
    egreso_id INT NOT NULL REFERENCES egreso(id),
    folio VARCHAR(100) NOT NULL,
    proveedor VARCHAR(150),
    fecha_emision DATE NOT NULL
);

-- ============================
--   INGRESOS
-- ============================

CREATE TABLE ingreso (
    id SERIAL PRIMARY KEY,
    fecha DATE NOT NULL,
    monto NUMERIC(12,2) NOT NULL,
    categoria_id INT NOT NULL REFERENCES categoria_ingreso(id),
    fondo_id INT REFERENCES fondo_inversion(id),
    descripcion TEXT
);

----------------------- Reportes que se deben de utilizar -------------------------------

CREATE VIEW vw_egresos_por_mes_categoria AS
SELECT
    pm.anio,
    pm.mes,
    ce.nombre AS categoria_egreso,
    SUM(e.monto) AS total_egresos
FROM egreso e
JOIN categoria_egreso ce ON e.categoria_id = ce.id
JOIN presupuesto_mensual pm ON e.presupuesto_id = pm.id
GROUP BY pm.anio, pm.mes, ce.nombre
ORDER BY pm.anio, pm.mes, ce.nombre;

------------------------Reporte por mes _ Fondos 

CREATE VIEW vw_ingresos_por_mes_categoria_fondo AS
SELECT
    EXTRACT(YEAR FROM i.fecha) AS anio,
    EXTRACT(MONTH FROM i.fecha) AS mes,
    ci.nombre AS categoria_ingreso,
    fi.nombre AS fondo,
    SUM(i.monto) AS total_ingresos
FROM ingreso i
JOIN categoria_ingreso ci ON i.categoria_id = ci.id
LEFT JOIN fondo_inversion fi ON i.fondo_id = fi.id
GROUP BY anio, mes, ci.nombre, fi.nombre
ORDER BY anio, mes, ci.nombre, fi.nombre;

-------------------------------------------------------- presupuesto vs egreso  

CREATE VIEW vw_presupuesto_vs_egresos AS
SELECT
    pm.id AS presupuesto_id,
    pm.anio,
    pm.mes,
    pm.monto AS presupuesto_mensual,
    COALESCE(SUM(e.monto), 0) AS total_egresos,
    pm.monto - COALESCE(SUM(e.monto), 0) AS saldo_disponible
FROM presupuesto_mensual pm
LEFT JOIN egreso e ON e.presupuesto_id = pm.id
GROUP BY pm.id, pm.anio, pm.mes, pm.monto
ORDER BY pm.anio, pm.mes;

------------------------------------------------------- ingresos mandados a cada fondo 

CREATE VIEW vw_ingresos_por_fondo AS
SELECT
    fi.id AS fondo_id,
    fi.nombre AS fondo,
    fi.tipo,
    SUM(i.monto) AS total_asignado
FROM fondo_inversion fi
LEFT JOIN ingreso i ON i.fondo_id = fi.id
GROUP BY fi.id, fi.nombre, fi.tipo
ORDER BY fi.nombre;

------------------------------------------------------- Auditoria de egresos con factura 

CREATE VIEW vw_egresos_facturados AS
SELECT
    e.id AS egreso_id,
    e.fecha,
    e.monto,
    ce.nombre AS categoria_egreso,
    pm.anio,
    pm.mes,
    f.id AS factura_id,
    f.folio,
    f.proveedor,
    f.fecha_emision
FROM egreso e
JOIN categoria_egreso ce ON e.categoria_id = ce.id
JOIN presupuesto_mensual pm ON e.presupuesto_id = pm.id
LEFT JOIN factura f ON f.egreso_id = e.id
ORDER BY e.fecha, e.id;

------------------------------------------------------------- function para evitar que  se exceda el presupuesto 

CREATE OR REPLACE FUNCTION fn_validar_presupuesto_egreso()
RETURNS TRIGGER AS $$
DECLARE
    v_presupuesto NUMERIC(12,2);
    v_total_egresos NUMERIC(12,2);
BEGIN
    SELECT monto INTO v_presupuesto
    FROM presupuesto_mensual
    WHERE id = NEW.presupuesto_id;

    SELECT COALESCE(SUM(monto), 0) INTO v_total_egresos
    FROM egreso
    WHERE presupuesto_id = NEW.presupuesto_id
      AND id <> COALESCE(NEW.id, -1);

    IF (v_total_egresos + NEW.monto) > v_presupuesto THEN
        RAISE EXCEPTION 'El egreso excede el presupuesto mensual (presupuesto: %, acumulado: %, nuevo egreso: %)',
            v_presupuesto, v_total_egresos, NEW.monto;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

------------------------------------------------------------  trigger sobre egreso 

CREATE TRIGGER trg_validar_presupuesto_egreso
BEFORE INSERT OR UPDATE ON egreso
FOR EACH ROW
EXECUTE FUNCTION fn_validar_presupuesto_egreso();

------------------------------------------------------------  para obtener el resumen financiero mensual  

CREATE OR REPLACE FUNCTION sp_resumen_financiero_mensual(p_anio INT, p_mes INT)
RETURNS TABLE (
    anio INT,
    mes INT,
    presupuesto NUMERIC(12,2),
    total_egresos NUMERIC(12,2),
    saldo_disponible NUMERIC(12,2),
    total_ingresos NUMERIC(12,2)
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        pm.anio,
        pm.mes,
        pm.monto AS presupuesto,
        COALESCE(SUM(DISTINCT e.monto), 0) AS total_egresos,
        pm.monto - COALESCE(SUM(DISTINCT e.monto), 0) AS saldo_disponible,
        COALESCE((
            SELECT SUM(i.monto)
            FROM ingreso i
            WHERE EXTRACT(YEAR FROM i.fecha) = p_anio
              AND EXTRACT(MONTH FROM i.fecha) = p_mes
        ), 0) AS total_ingresos
    FROM presupuesto_mensual pm
    LEFT JOIN egreso e ON e.presupuesto_id = pm.id
    WHERE pm.anio = p_anio
      AND pm.mes = p_mes
    GROUP BY pm.anio, pm.mes, pm.monto;
END;
$$ LANGUAGE plpgsql;

--------------------------------------------------------------- Listar movimientos de fondo mensual 
CREATE OR REPLACE FUNCTION sp_movimientos_fondo(p_fondo_id INT)
RETURNS TABLE (
    ingreso_id INT,
    fecha DATE,
    monto NUMERIC(12,2),
    categoria VARCHAR,
    fondo VARCHAR
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        i.id,
        i.fecha,
        i.monto,
        ci.nombre,
        fi.nombre
    FROM ingreso i
    JOIN categoria_ingreso ci ON i.categoria_id = ci.id
    JOIN fondo_inversion fi ON i.fondo_id = fi.id
    WHERE fi.id = p_fondo_id
    ORDER BY i.fecha;
END;
$$ LANGUAGE plpgsql;
